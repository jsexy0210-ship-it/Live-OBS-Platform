def addboard(name,title,x,y,w,h):
    c=json.load(open(P+'canvas.json',encoding='utf-8'))
    fn=name+'.dc.html'
    if fn not in c['boards']:
        c['order'].append(fn)
    old=c['boards'].get(fn)
    if old: x,y=old['x'],old['y']
    c['boards'][fn]={'x':x,'y':y,'w':w,'h':h,'title':title}
    json.dump(c,open(P+'canvas.json','w',encoding='utf-8'),ensure_ascii=False,indent=1)
